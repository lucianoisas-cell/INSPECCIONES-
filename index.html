/** * Backend para carga e inspección automática - Conexión Limpia */ function doGet(e) { // Si la app de GitHub intenta sincronizar (leer el historial): if (e && e.parameter.action === 'read') { var historial = getHistorial(); return ContentService.createTextOutput(JSON.stringify(historial)) .setMimeType(ContentService.MimeType.JSON); } // Si pide los datos del dashboard de vencimientos (hoja MAESTRO): if (e && e.parameter.action === 'maestro') { var maestro = getMaestro(); return ContentService.createTextOutput(JSON.stringify(maestro)) .setMimeType(ContentService.MimeType.JSON); } // Si no, devuelve la vista web normal (tu código original intacto): return HtmlService.createTemplateFromFile('Index') .evaluate() .setTitle('Control de Flota - Sistema') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } // Obtener registros reales para el historial de la web function getHistorial() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Hoja 1"); if (!sheet) return []; var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var headers = data[0]; var records = []; for (var i = 1; i < data.length; i++) { var row = data[i]; var record = { rowIndex: i + 1 }; for (var j = 0; j < headers.length; j++) { var headerName = headers[j]; var value = row[j]; if (value instanceof Date) { record[headerName] = Utilities.formatDate(value, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm"); } else { record[headerName] = value; } } records.push(record); } return records.reverse(); } // Obtener el listado completo de la flota (hoja MAESTRO) para el dashboard de vencimientos function getMaestro() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("MAESTRO"); if (!sheet) return []; var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var headers = data[0]; var records = []; for (var i = 1; i < data.length; i++) { var row = data[i]; // La columna INTERNO es la segunda (índice 1). Si está vacía, saltamos la fila. var interno = row[1]; if (interno === "" || interno === null || interno === undefined) continue; var record = {}; for (var j = 0; j < headers.length; j++) { var headerName = headers[j]; var value = row[j]; if (value instanceof Date) { record[headerName] = Utilities.formatDate(value, Session.getScriptTimeZone(), "dd/MM/yyyy"); } else { record[headerName] = value; } } records.push(record); } return records; } // Guardado e impacto automático en la planilla function guardarInspeccion(inspeccion) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet1 = ss.getSheetByName("Hoja 1"); var baseMaestra = ss.getSheetByName("BASE MAESTRA"); if (!sheet1 || !baseMaestra) return { success: false, error: "Error de configuración de hojas." }; var internoBuscado = parseInt(inspeccion.interno); if (isNaN(internoBuscado)) return { success: false, error: "Número de interno no válido." }; // Buscar datos maestros del vehículo por Interno var dataBase = baseMaestra.getDataRange().getValues(); var vehiculo = null; for (var i = 1; i < dataBase.length; i++) { if (parseInt(dataBase[i][0]) === internoBuscado) { vehiculo = { empresa: dataBase[i][1], patente: dataBase[i][2], marca: dataBase[i][3], modelo: dataBase[i][4], hab: dataBase[i][5], seguro: dataBase[i][6], vtv: dataBase[i][7], registro: dataBase[i][8] }; break; } } if (!vehiculo) { return { success: false, error: "El interno ingresado no existe en la BASE MAESTRA." }; } function formatFecha(val) { if (!val) return "---"; if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), "dd/MM/yyyy"); } return val.toString(); } var nuevaFila = [ Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss"), vehiculo.empresa, internoBuscado, vehiculo.patente, vehiculo.marca, vehiculo.modelo, inspeccion.chofer, inspeccion.inspector, inspeccion.resultado, inspeccion.observaciones, inspeccion.indumentaria, formatFecha(vehiculo.hab), formatFecha(vehiculo.seguro), formatFecha(vehiculo.vtv), formatFecha(vehiculo.registro) ]; sheet1.appendRow(nuevaFila); return { success: true }; } // Función para eliminar registros desde el celular/web function eliminarRegistro(rowIndex) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Hoja 1"); if (!sheet) return { success: false, error: "No se encontró la hoja." }; try { sheet.deleteRow(rowIndex); return { success: true }; } catch(e) { return { success: false, error: e.toString() }; } }
